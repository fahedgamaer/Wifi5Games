using UnityEngine; public class CarController : MonoBehaviour { [Header("Movement")] public float motorForce = 1800f; public float brakeForce = 2200f; public float maxSpeed = 18f; public float steeringRange = 35f; public float steeringSpeed = 8f; public float grip = 4f; public float downforce = 60f; [Header("Setup")] public Transform centerOfMass; public bool useKeyboard = true; private Rigidbody rb; private float steerInput; private float throttleInput; private float brakeInput; private float currentSteer; void Start() { rb = GetComponent(); if (centerOfMass != null) rb.centerOfMass = centerOfMass.localPosition; } void Update() { if (useKeyboard) { float horizontal = Input.GetAxisRaw("Horizontal"); float vertical = Input.GetAxisRaw("Vertical"); float brake = Input.GetKey(KeyCode.Space) ? 1f : 0f; SetInput(horizontal, vertical, brake); } } public void SetInput(float steer, float throttle, float brake) { steerInput = Mathf.Clamp(steer, -1f, 1f); throttleInput = Mathf.Clamp(throttle, -1f, 1f); brakeInput = Mathf.Clamp(brake, 0f, 1f); } void FixedUpdate() { // ثبات السيارة على الأرض rb.AddForce(-transform.up * downforce * rb.velocity.magnitude * 0.05f, ForceMode.Acceleration); // دوران السيارة float targetSteer = steerInput * steeringRange; currentSteer = Mathf.Lerp(currentSteer, targetSteer, Time.fixedDeltaTime * steeringSpeed); Quaternion turn = Quaternion.Euler(0f, currentSteer * Time.fixedDeltaTime * 60f, 0f); rb.MoveRotation(rb.rotation * turn); // السرعة الحالية على الاتجاه float speedForward = Vector3.Dot(rb.velocity, transform.forward); // الشحن / التباطؤ if (throttleInput > 0f && speedForward < maxSpeed) { rb.AddForce(transform.forward * throttleInput * motorForce, ForceMode.Acceleration); } if (throttleInput < 0f && speedForward > -maxSpeed * 0.7f) { rb.AddForce(transform.forward * throttleInput * motorForce * 0.7f, ForceMode.Acceleration); } // فرامل if (brakeInput > 0f) { Vector3 brakeVelocity = rb.velocity; brakeVelocity *= 0.92f; rb.velocity = Vector3.Lerp(rb.velocity, brakeVelocity, brakeInput * Time.fixedDeltaTime * 12f); } // تقليل الانزلاق float steerFactor = Mathf.Abs(steerInput); Vector3 forwardVelocity = Vector3.Project(rb.velocity, transform.forward); Vector3 sideVelocity = Vector3.Project(rb.velocity, transform.right); sideVelocity *= (1f - Mathf.Clamp01(steerFactor * 0.5f)); rb.velocity = forwardVelocity + sideVelocity; // تعطيل السرعة الزائدة if (rb.velocity.magnitude > maxSpeed * 1.5f) { rb.velocity = rb.velocity.normalized * maxSpeed * 1.5f; } } }
