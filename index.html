RegisterCommand("car", function(source, args, raw) local model = args[1] or "banshee" local playerPed = PlayerPedId() local pos = GetEntityCoords(playerPed) RequestModel(GetHashKey(model)) while not HasModelLoaded(GetHashKey(model)) do Wait(10) end local veh = CreateVehicle(GetHashKey(model), pos.x+5, pos.y, pos.z+1, GetEntityHeading(playerPed), true, false) SetPedIntoVehicle(playerPed, veh, -1) SetEntityAsNoLongerNeeded(veh) end, false) RegisterCommand("tpto", function(source, args, raw) local x,y,z = tonumber(args[1]), tonumber(args[2]), tonumber(args[3]) if x and y and z then local playerPed = PlayerPedId() SetEntityCoords(playerPed, x, y, z, false, false, false, true) TriggerEvent('chat:addMessage',{ args = {"^2Teleport","تم النقل"}}) else TriggerEvent('chat:addMessage',{ args = {"^1Usage","/tpto x y z"}}) end end, false)
