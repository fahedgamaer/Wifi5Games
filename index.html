using System; using GTA; using GTA.Native; public class Main : Script { public Main() { Tick += OnTick; UI.Notify("~g~GTA V Script Loaded"); } private void OnTick(object sender, EventArgs e) { if (Game.IsKeyPressed(System.Windows.Forms.Keys.F1)) { var player = Game.Player.Character; player.Position = new Vector3(0f, 0f, 30f); UI.Notify("~y~Teleported to spawn"); } if (Game.IsKeyPressed(System.Windows.Forms.Keys.F2)) { var vehicle = World.CreateVehicle(VehicleHash.Banshee, Game.Player.Character.Position + new Vector3(5f, 0f, 1f), 0f); if (vehicle != null) { vehicle.PlaceOnGround(); UI.Notify("~b~Spawned Banshee"); } } } }
