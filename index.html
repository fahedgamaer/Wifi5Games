create table public.listings ( id bigint generated by default as identity primary key, title text not null, description text, price numeric default 0, image_url text not null, created_at timestamptz default now() ); alter table public.listings enable row level security; create policy "Everyone can view listings" on public.listings for select to anon, authenticated using (true); create policy "Everyone can publish listings" on public.listings for insert to anon, authenticated with check (true); insert into storage.buckets (id, name, public) values ('listing-images', 'listing-images', true) on conflict (id) do nothing; create policy "Public can view listing images" on storage.objects for select to anon, authenticated using (bucket_id = 'listing-images'); create policy "Public can upload listing images" on storage.objects for insert to anon, authenticated with check (bucket_id = 'listing-images');
